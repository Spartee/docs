[Arcade](https://arcade.dev) is the leading MCP runtime that enables AI agents to take real-world actions securely at scale. It handles OAuth automatically, manages user tokens securely, and provides access to 1000s of tools across Gmail, Slack, GitHub, Google Sheets, Notion, Stripe, and many more services. As a native MCP server, Arcade works seamlessly with any MCP-compatible client including LangChain, LangGraph, Claude Desktop, Cursor, and VS Code.

## Instantiation

Initialize the Arcade client and fetch tools from specific toolkits or MCP servers. Each toolkit represents a service (e.g., Gmail, Slack) with multiple tools (actions you can perform).

```python Initialize Arcade icon="robot"
import asyncio
from arcadepy import AsyncArcade

# Initialize Arcade client
arcade = AsyncArcade()

# Get tools from specific toolkits
tools = await arcade.tools.list(toolkit="Gmail", limit=30)

print(f"Loaded {len(tools.items)} tools from Gmail toolkit")
```

## Converting Arcade tools to LangChain

Arcade tools need to be converted to LangChain-compatible @[StructuredTool] objects. Here's a complete conversion utility:

```python Type mapping
from typing import Any, List
from pydantic import BaseModel, Field, create_model
from langchain_core.tools import StructuredTool
from langchain_core.runnables import RunnableConfig
from arcadepy import AsyncArcade
from arcadepy.types import ToolDefinition

TYPE_MAPPING = {
    "string": str,
    "number": float,
    "integer": int,
    "boolean": bool,
    "array": list,
    "json": dict,
}

def get_python_type(val_type: str) -> Any:
    _type = TYPE_MAPPING.get(val_type)
    if _type is None:
        raise ValueError(f"Invalid value type: {val_type}")
    return _type

def arcade_schema_to_pydantic(tool_def: ToolDefinition) -> type[BaseModel]:
    fields: dict[str, Any] = {}
    for param in tool_def.input.parameters or []:
        param_type = get_python_type(param.value_schema.val_type)
        if param_type is list and param.value_schema.inner_val_type:
            inner_type = get_python_type(param.value_schema.inner_val_type)
            param_type = list[inner_type]
        param_description = param.description or "No description"
        default = ... if param.required else None
        fields[param.name] = (
            param_type,
            Field(default=default, description=param_description),
        )
    return create_model(f"{tool_def.name}Args", **fields)
```

```python Convert Arcade tools to LangChain
async def arcade_to_langchain(
    arcade_client: AsyncArcade,
    arcade_tool: ToolDefinition,
) -> StructuredTool:
    args_schema = arcade_schema_to_pydantic(arcade_tool)

    async def tool_function(config: RunnableConfig, **kwargs: Any) -> Any:
        user_id = config.get("configurable", {}).get("user_id") if config else None
        if not user_id:
            raise ValueError("User ID required for Arcade tools")

        # Check authorization status
        auth_response = await arcade_client.tools.authorize(
            tool_name=arcade_tool.qualified_name,
            user_id=user_id
        )

        if auth_response.status != "completed":
            # Return authorization URL for user to complete
            return {
                "authorization_required": True,
                "auth_url": auth_response.url,
                "auth_id": auth_response.id
            }

        # Execute the tool
        filtered_kwargs = {k: v for k, v in kwargs.items() if v is not None}
        response = await arcade_client.tools.execute(
            tool_name=arcade_tool.qualified_name,
            input=filtered_kwargs,
            user_id=user_id,
        )

        if response.output and response.output.value:
            return response.output.value

        return {"error": "Tool execution failed"}

    return StructuredTool.from_function(
        coroutine=tool_function,
        name=arcade_tool.qualified_name.replace(".", "_"),
        description=arcade_tool.description,
        args_schema=args_schema
    )
```

## Get tools from Arcade

Retrieve tools from specific toolkits or by name:

```python Get tools helper
async def get_arcade_tools(
    arcade_client: AsyncArcade | None = None,
    toolkits: List[str] | None = None,
    tools: List[str] | None = None,
    tool_limit: int = 30,
) -> List[StructuredTool]:
    if not arcade_client:
        arcade_client = AsyncArcade()

    if not tools and not toolkits:
        raise ValueError("Provide either tools or toolkits")

    tool_definitions: dict[str, ToolDefinition] = {}

    # Get specific tools by name
    if tools:
        tasks = [arcade_client.tools.get(name=tool_name) for tool_name in tools]
        responses = await asyncio.gather(*tasks)
        for response in responses:
            tool_definitions[response.fully_qualified_name] = response

    # Get all tools from toolkits
    if toolkits:
        tasks = [
            arcade_client.tools.list(toolkit=toolkit, limit=tool_limit)
            for toolkit in toolkits
        ]
        responses = await asyncio.gather(*tasks)
        for response in responses:
            for tool in response.items:
                tool_definitions[tool.fully_qualified_name] = tool

    # Convert to LangChain tools
    tasks = [
        arcade_to_langchain(arcade_client, tool_def)
        for tool_def in tool_definitions.values()
    ]
    return await asyncio.gather(*tasks)
```

```python Agent with Arcade tools icon="robot"
import asyncio
from arcadepy import AsyncArcade
from langchain_openai import ChatOpenAI
from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver

async def main():
    # Initialize Arcade
    arcade = AsyncArcade()

    # Get Gmail tools
    tools = await get_arcade_tools(
        arcade_client=arcade,
        tools=["Gmail_ListEmails", "Gmail_SendEmail"]
    )

    # Create model
    model = ChatOpenAI(model="gpt-4o")

    # Create agent with memory
    memory = MemorySaver()
    agent = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    # Configure with user ID for authorization
    config = {
        "configurable": {
            "thread_id": "email_thread",
            "user_id": "user@example.com"
        }
    }

    # Run the agent
    result = await agent.ainvoke(
        {"messages": [{"role": "user", "content": "List my latest 3 emails"}]},
        config=config
    )

    print(result["messages"][-1].content)

asyncio.run(main())
```

## Authentication handling

Arcade manages OAuth flows for you. When a tool requires authorization, you'll receive an auth URL:

```python
from arcadepy import Arcade

arcade = Arcade()
user_id = "user@example.com"

# Start authorization for a provider
auth_response = arcade.auth.start(
    user_id=user_id,
    provider="google",
    scopes=["https://www.googleapis.com/auth/gmail.readonly"],
)

if auth_response.status != "completed":
    print("Please authorize the application:")
    print(auth_response.url)

# Wait for user to complete authorization
auth_response = arcade.auth.wait_for_completion(auth_response)
print("Authorization completed!")
```

## Using with LangChain tools directly

You can also use Arcade to get OAuth tokens and use them with existing LangChain tool integrations:

```python
from arcadepy import Arcade
from google.oauth2.credentials import Credentials
from langchain_google_community import GmailToolkit
from langchain_google_community.gmail.utils import build_resource_service

# Get token from Arcade
arcade = Arcade()
auth_response = arcade.auth.start(
    user_id="user@example.com",
    provider="google",
    scopes=["https://www.googleapis.com/auth/gmail.readonly"],
)

if auth_response.status != "completed":
    print(f"Authorize at: {auth_response.url}")
    auth_response = arcade.auth.wait_for_completion(auth_response)

# Use token with LangChain Gmail toolkit
creds = Credentials(auth_response.context.token)
api_resource = build_resource_service(credentials=creds)
toolkit = GmailToolkit(api_resource=api_resource)
tools = toolkit.get_tools()
```

## Multi-user scenarios

Arcade handles per-user authentication automatically:

```python
# Each user has their own authenticated sessions
tools_user_1 = await get_arcade_tools(
    toolkits=["Gmail"],
    user_id="user_1@example.com"
)
tools_user_2 = await get_arcade_tools(
    toolkits=["Gmail"],
    user_id="user_2@example.com"
)

# Tools will use the respective user's credentials
agent_1 = create_react_agent(model, tools_user_1)
agent_2 = create_react_agent(model, tools_user_2)
```

### Custom tools

Create custom tools using the Arcade SDK:

```python
from arcadepy import Arcade

arcade = Arcade()

# Register a custom tool
custom_tool = arcade.tools.define(
    name="MyCustomTool",
    description="A custom tool for my application",
    input_schema={
        "type": "object",
        "properties": {
            "query": {"type": "string", "description": "Search query"}
        },
        "required": ["query"]
    }
)
```
